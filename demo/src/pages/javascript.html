السمات

كلّ ما يلي يعمل بالسمة وحدها في الشيفرة، دون سطر جافاسكربت في صفحتك. هذا هو الباب الأمامي؛ الواجهة البرمجية تحته مخرج طوارئ لما لا تصل إليه سمة.

السمةعلىما تفعله
data-ins-theme-toggleزرّيبدّل بين الفاتح والداكن. بالقيمة system يعيد اتّباع النظام.
data-ins-fxزرّيخفّف المؤثرات أو يعيدها.
data-ins-sidebarزرّيطوي الشريط الجانبي، أو يفتح الدرج على الهاتف.
data-ins-dialog="#id"زرّيفتح حوارًا أو درجًا.
data-ins-dialog-closeزرّيغلق الحوار الذي هو فيه.
data-ins-dismissزرّيغلق أقرب ما يمكن إغلاقه: تنبيه، حوار، درج. أو ما تسمّيه بقيمة #id.
data-ins-confirm="نص"زرّ، رابط، نموذجيسأل قبل أن يمضي الإجراء. زرّ الموافقة بـ data-ins-confirm-ok، والخطر بـ data-ins-confirm-tone="danger".
data-ins-toast="نص"زرّيُظهر رسالة طائرة. النبرة بـ data-ins-tone.
data-ins-tab="#id".ins-tabيبدّل لوح الألسنة.
data-ins-tipأيّ عنصرتلميح. فارغًا يعيد استعمال aria-label. الجهة بـ data-ins-tip-side.
data-ins-toggleزرّزرّ تبديل يحفظ حالته في aria-pressed.
data-ins-navbar.ins-navbar-toggleيفتح قائمة الموقع على الهاتف.
data-ins-password.ins-password-toggleيُظهر كلمة المرور ويخفيها.
data-ins-validate<form>تحقّق المتصفّح برسائل داخل الحقل.
data-ins-dateinput[type=date]منتقي تاريخ بلغة الصفحة. للمدى: data-ins-date-start و-end.
data-ins-spin="up|down"زرّيزيد الحقل الرقمي المجاور أو ينقصه.
data-ins-indeterminateمربّع اختيارحالة «بعضها محدّد».
data-ins-wizard="next|prev"زرّ داخل .ins-wizardينتقل بين خطوات المعالج بعد التحقّق.
data-ins-revealحاويةيدخل أبناؤها متدرّجين.

على عنصر <html>

والسكربت يختم data-ins-js على <html> بنفسه قبل أوّل رسم. كلّ حالة تحتاج السكربت معلّقة بهذه السمة — لوح لسان مخفيّ، قائمة موقع مطويّة — فإن فشل السكربت ظهر كلّ شيء بدل أن يختفي.

الواجهة البرمجية

Insiyab.version // '0.3.0' Insiyab.theme() // 'light' | 'dark' — ما يراه المستخدم فعلًا Insiyab.theme('dark' | 'light' | 'system') Insiyab.toggleTheme() Insiyab.brand('#0F766E') // يشتقّ لون الهوية ويطبّقه Insiyab.sidebar('open' | 'collapsed' | 'toggle') Insiyab.toast('تمّ الحفظ', 'ok') // 'ok' | 'bad' | 'warn' | 'info' Insiyab.dialog('#invite', 'open') // 'open' | 'close' | بلا قيمة للتبديل Insiyab.confirm('حذف الملفّ؟', { title: 'تأكيد', confirm: 'حذف', cancel: 'تراجع', tone: 'danger' }) .then((yes) => { /* true | false */ }) Insiyab.dismiss(el) // يغلق ما يحويه، كما تفعل data-ins-dismiss Insiyab.tab('#tab-billing') Insiyab.navbar(el, true) Insiyab.wizard('#signup', 2) // ينتقل إلى الخطوة الثالثة Insiyab.date('#from', '2026-10-01') Insiyab.init(scope) // يبني المكوّنات داخل عناصر أضفتها أنت Insiyab.define('chart', (scope) => {}) // يضيف منشئًا خاصًّا بك إلى ذلك المسح Insiyab.scrollTop() // موضع التمرير، أيًّا كان العنصر المتمرِّر Insiyab.color.derive('#0F766E', 'dark') // المجموعة كاملة دون تطبيقها Insiyab.color.contrast('#fff', '#0F766E') Insiyab.color.labelFor('#C9A063') // '#111111' — مقيس Insiyab.color.shift('#0F766E', -0.3)

الأحداث

كلّ حدث يُطلَق على document ومعه detail. جرّب الأزرار: السجلّ تحتها يستمع إلى الأحداث الحقيقية ويكتب ما يصله.

قائمة 
// الأحداث تظهر هنا
 document.addEventListener('ins:theme', (e) => e.detail.theme) // 'dark' document.addEventListener('ins:seg', (e) => e.detail.value) // 'week' document.addEventListener('ins:toggle', (e) => e.detail.pressed) // true document.addEventListener('ins:menu', (e) => e.detail.checked) document.addEventListener('ins:tab', (e) => e.detail.panel) document.addEventListener('ins:sidebar', (e) => e.detail.state) // 'collapsed' document.addEventListener('ins:combo', (e) => e.detail.value) document.addEventListener('ins:date', (e) => e.detail.value) // '2026-10-01' document.addEventListener('ins:wizard', (e) => e.detail.step) document.addEventListener('ins:navbar', (e) => e.detail.open) document.addEventListener('ins:dismiss', (e) => e.detail.target)

منشئ خاصّ بك

Insiyab.define يسجّل دالّة تُستدعى في مسح التهيئة نفسه — عند التحميل، ومع كلّ Insiyab.init(scope). فمكوّنك يُبنى حيث تُبنى مكوّنات المكتبة، ولا تحتاج نظامًا ثانيًا.

Insiyab.define('sparkline', (scope) => { scope.querySelectorAll('[data-sparkline]').forEach((el) => { if (el.dataset.ready) return; // المسح قد يتكرّر على العنصر نفسه el.dataset.ready = '1'; drawSparkline(el, el.dataset.sparkline.split(',').map(Number)); }); });
