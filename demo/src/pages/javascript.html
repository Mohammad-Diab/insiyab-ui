السمات

كل ما في هذا الجدول يعمل بالسمة وحدها، فلا تكتب في صفحتك سطرًا من جافاسكربت.

السمةالعنصرعملها
data-ins-theme-toggleزريبدل بين الفاتح والداكن، وبالقيمة system يعود إلى اتباع النظام.
data-ins-fxزريخفف المؤثرات أو يعيدها.
data-ins-sidebarزريطوي الشريط الجانبي، وعلى الهاتف يفتح الدرج.
data-ins-dialog="#id"زريفتح مربع حوار أو درجًا.
data-ins-dialog-closeزريغلق الحوار الذي يضمه.
data-ins-dismissزريغلق أقرب ما يقبل الإغلاق من تنبيه أو حوار أو درج، أو ما تسميه بقيمة #id.
data-ins-confirm="نص"زر أو رابط أو نموذجيسأل المستخدم قبل أن يمضي الإجراء. وتضع نص زر الموافقة في data-ins-confirm-ok، وتنبه إلى الخطر بـ data-ins-confirm-tone="danger".
data-ins-toast="نص"زريعرض رسالة عابرة، وتختار نبرتها بـ data-ins-tone.
data-ins-tab="#id".ins-tabيعرض لوح التبويب الذي يشير إليه.
data-ins-tipأي عنصريضع تلميحًا، وإذا تركت السمة فارغة أخذ نصه من aria-label. وتحدد جهته بـ data-ins-tip-side.
data-ins-toggleزريجعل الزر مفتاح تبديل يحفظ حالته في aria-pressed.
data-ins-navbar.ins-navbar-toggleيفتح قائمة الموقع على الهاتف.
data-ins-password.ins-password-toggleيكشف كلمة المرور ويخفيها.
data-ins-validate<form>يجري تحقق المتصفح، ويعرض رسائله داخل الحقل نفسه.
data-ins-dateinput[type=date]يبني منتقي تاريخ بلغة الصفحة، ولاختيار فترة بين تاريخين تستعمل data-ins-date-start و-end.
data-ins-spin="up|down"زريزيد الحقل الرقمي المجاور أو ينقصه.
data-ins-indeterminateمربع اختياريضع المربع في حالة «بعضها محدد».
data-ins-wizard="next|prev"زر داخل .ins-wizardينتقل بين خطوات المعالج بعد التحقق.
data-ins-revealحاويةتدخل عناصرها متتابعة.

على عنصر <html>

ويضع السكربت data-ins-js على <html> بنفسه قبل أن تظهر الصفحة، وكل حالة تحتاج السكربت مشروطة بهذه السمة، فإن تعطل السكربت ظهر كل شيء ولم يختف منه شيء.

الواجهة البرمجية

Insiyab.version // '0.6.0' Insiyab.theme() // 'light' | 'dark' — ما يراه المستخدم فعلًا Insiyab.theme('dark' | 'light' | 'system') Insiyab.toggleTheme() Insiyab.brand('#0F766E') // يشتق لون الهوية ويطبقه Insiyab.sidebar('open' | 'collapsed' | 'toggle') Insiyab.toast('تم الحفظ', 'ok') // 'ok' | 'bad' | 'warn' | 'info' Insiyab.dialog('#invite', 'open') // 'open' | 'close' | بلا قيمة للتبديل Insiyab.confirm('حذف الملف؟', { title: 'تأكيد', confirm: 'حذف', cancel: 'تراجع', tone: 'danger' }) .then((yes) => { /* true | false */ }) Insiyab.dismiss(el) // يغلق ما يحويه، كما تفعل data-ins-dismiss Insiyab.tab('#tab-billing') Insiyab.navbar(el, true) Insiyab.wizard('#signup', 2) // ينتقل إلى الخطوة الثالثة Insiyab.date('#from', '2026-10-01') Insiyab.init(scope) // يبني المكونات داخل عناصر أضفتها أنت Insiyab.define('chart', (scope) => {}) // يضيف دالة تهيئة خاصة بك تعمل مع init Insiyab.scrollTop() // موضع التمرير، أيًا كان العنصر المتمرر Insiyab.color.derive('#0F766E', 'dark') // المجموعة كاملة دون تطبيقها Insiyab.color.contrast('#fff', '#0F766E') Insiyab.color.labelFor('#C9A063') // '#111111' — يختاره بحساب التباين Insiyab.color.shift('#0F766E', -0.3)

الأحداث

تُطلق الأحداث كلها على document، ويحمل كل منها تفاصيله في detail. جرب الأزرار، وسيكتب السجل تحتها ما يصله.

// الأحداث تظهر هنا
 document.addEventListener('ins:theme', (e) => e.detail.theme) // 'dark' document.addEventListener('ins:seg', (e) => e.detail.value) // 'week' document.addEventListener('ins:toggle', (e) => e.detail.pressed) // true document.addEventListener('ins:menu', (e) => e.detail.checked) document.addEventListener('ins:tab', (e) => e.detail.panel) document.addEventListener('ins:sidebar', (e) => e.detail.state) // 'collapsed' document.addEventListener('ins:combo', (e) => e.detail.value) document.addEventListener('ins:date', (e) => e.detail.value) // '2026-10-01' document.addEventListener('ins:wizard', (e) => e.detail.step) document.addEventListener('ins:navbar', (e) => e.detail.open) document.addEventListener('ins:dismiss', (e) => e.detail.target)

دالة تهيئة خاصة بك

تسجل Insiyab.define دالة تستدعيها المكتبة كلما هيأت المكونات، عند التحميل ومع كل Insiyab.init(scope).

Insiyab.define('sparkline', (scope) => { scope.querySelectorAll('[data-sparkline]').forEach((el) => { if (el.dataset.ready) return; // قد تمر التهيئة على العنصر نفسه مرتين el.dataset.ready = '1'; drawSparkline(el, el.dataset.sparkline.split(',').map(Number)); }); });
