Values are always escaped: an attribute never holds markup. cls(list): the class names that are set, joined. -#} {% macro attrs(a) -%} {%- for k, v in a|dictsort if v is defined and v is not none and v is not sameas false -%} {{ ' ' }}{{ k|replace('_', '-') }}{% if v is not sameas true %}="{{ v|e }}"{% endif %} {%- endfor -%} {%- endmacro %} {% macro cls(list) -%} {{ list|select|join(' ') }} {%- endmacro %}
